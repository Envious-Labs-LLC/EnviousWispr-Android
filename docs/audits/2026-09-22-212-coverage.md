```kotlin
package com.envi.wispr.asr

import java.util.concurrent.ExecutorService
import java.util.concurrent.RejectedExecutionException
import java.util.concurrent.atomic.AtomicBoolean

internal class RecognizerOwner<R : Any>(
    private val worker: ExecutorService,
    private val free: (R) -> Unit,
) {
    private val closed = AtomicBoolean(false)

    /** Read and written only on [worker]. */
    private var recognizer: R? = null

    @Volatile
    private var ready = false

    /** A close permanently masks readiness, even if an in-flight load finishes afterward. */
    val isReady: Boolean
        get() = !closed.get() && ready

    fun load(open: () -> R?) = submit(refused = {}) {
        val opened = open()
        recognizer = opened
        ready = opened != null
    }

    fun use(refused: () -> Unit, work: (R?) -> Unit) =
        submit(refused) { work(recognizer) }

    fun close() {
        if (!closed.compareAndSet(false, true)) return
        ready = false
        try {
            worker.execute {
                ready = false
                val current = recognizer
                recognizer = null
                if (current != null) free(current)
            }
        } catch (_: RejectedExecutionException) {
            // The process-level shutdown policy owns recovery if release could not be queued.
        }
        worker.shutdown()
    }

    private fun submit(refused: () -> Unit, task: () -> Unit) {
        try {
            worker.execute {
                if (closed.get()) refused() else task()
            }
        } catch (_: RejectedExecutionException) {
            refused()
        }
    }
}
```
