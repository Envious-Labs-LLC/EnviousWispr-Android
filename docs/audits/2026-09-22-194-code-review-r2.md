   173	    /**
   174	     * The balanced-parenthesis argument text of every diagnostic call; a multi-line call is one span. A
   175	     * token may follow `.` (`this.log.warn(`, `com.envi.wispr.debug.DebugLogger.log(`) but not a word
   176	     * character (`catalog.warn(`). Parentheses and commas inside string and character literals are skipped
   177	     * by [scanArguments].
   178	     */
   179	    private fun diagnosticSpans(text: String): List<String> = buildList {
   180	        for (token in diagnosticCallTokens) {
   181	            var from = text.indexOf(token)
   182	            while (from >= 0) {
   183	                val before = if (from == 0) ' ' else text[from - 1]
   184	                if (!before.isLetterOrDigit() && before != '_') {
   185	                    add(scanArguments(text, from + token.length - 1).first)
   186	                }
   187	                from = text.indexOf(token, from + 1)
   188	            }
   189	        }
   190	    }
   191	
   192	    /**
   193	     * From the `(` at [open], returns the argument text up to its matching `)` and the count of top-level
   194	     * commas, treating `"…"` (with `\"` escapes), `"""…"""` and `'…'` (with `\'` escapes) as opaque.
   195	     */
   196	    private fun scanArguments(text: String, open: Int): Pair<String, Int> {
   197	        var depth = 0
   198	        var commas = 0
   199	        var i = open
   200	        while (i < text.length) {
   201	            val c = text[i]
   202	            when {
   203	                text.startsWith("\"\"\"", i) -> {
   204	                    val end = text.indexOf("\"\"\"", i + 3)
   205	                    i = if (end < 0) text.length else end + 3
   206	                    continue
   207	                }
   208	                c == '"' || c == '\'' -> {
   209	                    var j = i + 1
   210	                    while (j < text.length && text[j] != c) {
   211	                        if (text[j] == '\\') j++
   212	                        j++
   213	                    }
   214	                    i = j + 1
   215	                    continue
   216	                }
   217	                c == '(' -> depth++
   218	                c == ')' -> if (--depth == 0) return text.substring(open + 1, i) to commas
   219	                c == ',' -> if (depth == 1) commas++
   220	            }
   221	            i++
   222	        }
   223	        return text.substring(open + 1) to commas
   224	    }
