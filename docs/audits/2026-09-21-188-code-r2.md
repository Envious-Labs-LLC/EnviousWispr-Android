   375	                    throw IllegalStateException("AudioRecord failed to initialize")
   376	                }
   377	                val effective = EffectiveDevice(route.reason)
   378	                val takeRoute = TakeRoute(
   379	                    hold = routeHold,
   380	                    resolved = route,
   381	                    effective = effective,
   382	                    gate = LiveGate(gated = route.needsBluetooth),
   383	                    phonePicked = pick is InputDevicePick.Device && InputRouteKind.of(pick.type) == InputRouteKind.PHONE,
   384	                    listenerSlot = routingListener,
   385	                    scheduler = routeScheduler,
   386	                    unregisterDeviceCallback = { audioManager.unregisterAudioDeviceCallback(it) },
   387	                    tag = TAG,
   388	                )
   389	                takeRoute.applyPreferred(record)
   390	
   391	                // cacheDir is app-internal and shared by this package's processes. It is
   392	                // excluded from user backups and is less exposed than shared storage.
   393	                val file = File(cacheDir, AUDIO_FILENAME)
   394	                // There is no active session under sessionLock, so no writer can own this
   395	                // path. Remove a partial take left by process death before opening a new one.
   396	                if (file.exists() && !file.delete()) {
   397	                    throw IOException("Unable to remove stale audio recording")
   398	                }
   399	                output = FileOutputStream(file)
   400	                record.startRecording()
   401	                // The route is the target from the first read when the preferred device was set first
   402	                // (V2, 2026-09-16: routedDevice already the SCO source at startRecording).
   403	                record.routedDevice?.let { effective.observe(it.type, it.productName?.toString().orEmpty()) }
   404	                takeRoute.markRecorderStarted(SystemClock.elapsedRealtime())
   405	
   406	                val newSession = CaptureSession(
   407	                    record = record,
   408	                    file = file,
   409	                    output = output,
   410	                    // Allocated HERE, before the thread starts, and never inside the capture loop. The
   411	                    // capture thread may not allocate: it must do nothing that can make it late.
   412	                    readBuffer = ByteArray(PcmAudio.READ_CHUNK_BYTES),
   413	                    token = nextCaptureToken(),
   414	                    detector = DetectorFeed(
   415	                        tag = TAG,
   416	                        autoStop = detectorEnabled,
   417	                        bind = DetectorFeed.bindingThrough(this),
   418	                        unbind = DetectorFeed.unbindingThrough(this),
   419	                    ),
   420	                    picture = PicturePublisher(spectrumListener, TAG),
   421	                    route = takeRoute,
   422	                    keepEarbudsReady = keepEarbudsReady,
   423	                    takeId = takeId,
   424	                )
   425	                takeRoute.registerListener(record, routeHandler) { newSession.bytesWritten }
   426	                session = newSession
   427	                lastEffective = effective
   428	                lastAudioFile = file
   429	                isRecording.set(true)
   430	                terminalReason = TERMINAL_REASON_NONE
   431	                currentAmplitude = 0f
   432	                takePeakAmplitude = 0f
   433	                DebugLogger.startPipeline()
   434	                DebugLogger.mark(TAG, "recording_start")
   435	                DebugLogger.log(
   436	                    TAG,
   437	                    "Recording started (PID: ${android.os.Process.myPid()}, " +
   438	                        "max: ${RecordingLimits.MAX_DURATION_MS}ms, " +
   439	                        "nativeFrames: ${runCatching { record.bufferSizeInFrames }.getOrDefault(-1)})",
   440	                )
   441	                DebugLogger.log(
   442	                    TAG,
   443	                    "route start=${effective.label()} kind=${effective.kind} reason=${route.reason} " +
   444	                        "target=${route.target.label}",
   445	                )
   446	
   447	                if (requestedButRefused) {
   448	                    // The caller asked for auto-stop and cannot have it, which is exactly the state the
   449	                    // notice exists for. Recording itself is unaffected.
   450	                    newSession.detector.markRequestedButRefused()
   451	                    DebugLogger.warn(TAG, "Auto-stop refused: pause $pauseSeconds is out of range")
   452	                }
   453	                if (detectorEnabled) {
   454	                    newSession.detector.start(
   455	                        pauseSeconds = validPause!!,
   456	                        token = newSession.token,
   457	                        takeId = newSession.takeId,
   458	                        isCurrent = { session === newSession },
   459	                        stillLive = { session === newSession && !newSession.stopRequested },
   460	                        endOnSilence = { endTake(newSession, TERMINAL_REASON_SILENCE) },
   461	                    )
   462	                }
   463	
   464	                val thread = Thread({ captureLoop(newSession) }, "AudioCaptureThread")
   465	                captureThread = thread
   466	                try {
   467	                    thread.start()
   468	                } catch (e: Exception) {
   469	                    isRecording.set(false)
   470	                    session = null
   471	                    closeResources(newSession, keepRoute = false)
   472	                    captureThread = null
   473	                    stopSelf()
   474	                    DebugLogger.error(TAG, "Failed to start capture thread", e)
   475	                    return false
   476	                }
   477	                newSession.picture.start(stillLive = { session === newSession && !newSession.stopRequested })
   478	                if (route.needsBluetooth) {
   479	                    takeRoute.watchSink(audioManager, routeHandler)
   480	                    takeRoute.armDeadline(
   481	                        audioManager = audioManager,
   482	                        locked = { block -> synchronized(sessionLock) { block() } },
   483	                        stillWaiting = { session === newSession && isRecording.get() && takeRoute.gate.state == LiveGate.State.WAITING },
   484	                        onRefused = {
   485	                            lastStartFailure = START_FAILURE_EARBUDS
   486	                            endTakeLocked(newSession, TERMINAL_REASON_ERROR)
   487	                        },
   488	                    )
   489	                } else {
   490	                    takeRoute.markLive()
   491	                }
   492	                return thread.isAlive && session === newSession && isRecording.get()
   493	            } catch (e: SecurityException) {
   494	                DebugLogger.error(TAG, "RECORD_AUDIO permission not granted", e)
   495	                lastStartFailure = START_FAILURE_OTHER
   496	                isRecording.set(false)
   497	                routeHold.release()
   498	                closeResources(record, output)
   499	                stopSelf()
   500	                return false
   501	            } catch (e: Exception) {
   502	                DebugLogger.error(TAG, "Failed to start recording", e)
   503	                lastStartFailure = START_FAILURE_OTHER
   504	                isRecording.set(false)
   505	                routeHold.release()
   506	                closeResources(record, output)
   507	                stopSelf()
   508	                return false
   509	            }
   510	        }
   511	    }

   513	    private fun captureLoop(active: CaptureSession) {
   514	        val buffer = active.readBuffer
   515	        try {
   516	            while (isRecording.get() && session === active) {
   517	                // The cap counts from LIVE; the wait for the earbuds has its own bound in the session owner.
   518	                val live = active.route.liveAtMs
   519	                val elapsed = if (live > 0L) SystemClock.elapsedRealtime() - live else 0L
   520	                if (elapsed >= RecordingLimits.MAX_DURATION_MS) {
   521	                    // The reason is the whole signal. The session owner reads it back through
   522	                    // getTerminalReason and is what tells the user why their take ended.
   523	                    claimEnding(active, TERMINAL_REASON_MAX_DURATION)
   524	                    DebugLogger.log(TAG, "Max duration reached (${elapsed}ms), auto-stopping")
   525	                    break
   526	                }
   527	
   528	                // THE CLOCK CANNOT BOUND THE FILE, because the two measure different things.
   529	                // `record.startRecording()` runs before `startedAtMs` is taken, so the hardware is
   530	                // already buffering when the clock starts, and the check above can pass on its last
   531	                // pass with the file already at the ceiling. One more whole block then goes in.
   532	                //
   533	                // That is not a rounding error. `AsrService` REFUSES a file over the ceiling rather
   534	                // than truncating it, so a few bytes past it discards the entire take: exactly the
   535	                // long dictation this limit exists to keep. So the read is bounded by what is left.
   536	                val remaining = RecordingLimits.MAX_AUDIO_BYTES - active.bytesWritten
   537	                if (remaining <= 0L) {
   538	                    claimEnding(active, TERMINAL_REASON_MAX_DURATION)
   539	                    DebugLogger.log(TAG, "Byte ceiling reached (${active.bytesWritten} bytes), auto-stopping")
   540	                    break
   541	                }
   542	
   543	                // Explicit, because the three-argument overload's blocking behaviour is a default
   544	                // rather than a statement, and this loop's timing depends on it.
   545	                val requested = minOf(buffer.size.toLong(), remaining).toInt()
   546	                val bytesRead = active.record.read(buffer, 0, requested, AudioRecord.READ_BLOCKING)
   547	                if (bytesRead < 0) throw IOException("AudioRecord.read failed: $bytesRead")
   548	                if (bytesRead == 0) continue
   549	
   550	                // Until the gate opens, a read feeds the gate and nothing else: not the file, not the
   551	                // detector, not the picture, not the level. The take's clock starts when the gate opens.
   552	                if (active.route.gate.state == LiveGate.State.WAITING) {
   553	                    val admissible = active.route.admissible()
   554	                    if (active.route.gate.offer(buffer, bytesRead, admissible)) active.route.markLive() else continue
   555	                }
   556	
   557	                val position = active.bytesWritten
   558	                active.output.write(buffer, 0, bytesRead)
   559	                active.bytesWritten += bytesRead
   560	                active.liveVisible = true
   561	                active.detector.offer(buffer, bytesRead, position)
   562	                active.picture.offer(buffer, bytesRead, position)
   563	
   564	                var sum = 0L
   565	                var peak = 0
   566	                for (i in 0 until bytesRead step PcmAudio.BYTES_PER_SAMPLE) {
   567	                    if (i + 1 < bytesRead) {
   568	                        val sample = (buffer[i].toInt() and 0xFF) or
   569	                            (buffer[i + 1].toInt() shl 8)
   570	                        val magnitude = abs(sample)
   571	                        sum += magnitude
   572	                        if (magnitude > peak) peak = magnitude
   573	                    }
   574	                }
   575	                val numSamples = bytesRead / PcmAudio.BYTES_PER_SAMPLE
   576	                currentAmplitude = if (numSamples > 0) {
   577	                    (sum.toFloat() / numSamples) / Short.MAX_VALUE
   578	                } else 0f
   579	                // No allocation, one compare per read: the take's peak, kept for the reader at stop.
   580	                val peakLevel = peak.toFloat() / Short.MAX_VALUE
   581	                if (peakLevel > takePeakAmplitude) takePeakAmplitude = peakLevel
   582	            }
   583	        } catch (e: Exception) {
   584	            synchronized(sessionLock) {
   585	                if (session === active) claimEnding(active, TERMINAL_REASON_ERROR)
   586	            }
   587	            DebugLogger.error(TAG, "Capture thread error", e)
   588	        } finally {
   589	            releaseSession(active)
   590	        }
   591	    }

   593	    /**
   594	     * Publish the first ending claimed for [active] and stop the loop.
   595	     *
   596	     * The published reason must outlive [releaseSession], because the client polls `getTerminalReason`
   597	     * only AFTER `isCapturing` has gone false, by which point the session is gone. So the value lives on
   598	     * the service while the CLAIM lives on the session, and the claim is what makes it first-wins.
   599	     */
   600	    private fun claimEnding(active: CaptureSession, reason: Int): Boolean {
   601	        if (!active.endingClaim.claim(reason)) return false
   602	        terminalReason = reason
   603	        isRecording.set(false)
   604	        return true
   605	    }

   607	    /** The user, or anything else with the authority to end whatever take is running. */
   608	    private fun stopRecording() {
   609	        synchronized(sessionLock) {
   610	            val active = session ?: return
   611	            if (!isRecording.get()) return
   612	            endTakeLocked(active, TERMINAL_REASON_MANUAL)
   613	        }
   614	    }
   615	
   616	    /**
   617	     * End exactly [expected], and nothing else.
   618	     *
   619	     * The identity check and the claim happen under ONE hold of the lock. Checking outside it and then
   620	     * ending "the current session" is the shape that lets a detector result from a take that has already
   621	     * finished stop the recording that started after it.
   622	     */
   623	    private fun endTake(expected: CaptureSession, reason: Int) {
   624	        synchronized(sessionLock) {
   625	            if (session !== expected || !isRecording.get()) return
   626	            endTakeLocked(expected, reason)
   627	        }
   628	    }
   629	
   630	    /**
   631	     * Signal the reader to finish. The reader performs the one final native release.
   632	     *
   633	     * Called only while [sessionLock] is held and [active] is still the current session.
   634	     */
   635	    private fun endTakeLocked(active: CaptureSession, reason: Int) {
   636	        if (!claimEnding(active, reason)) return
   637	        // routedDevice returns null once the recorder is inactive, so the final route is read HERE,
   638	        // before stop(); a headset removed just before the stop is then in the record even when its
   639	        // routing callback runs late. A null read preserves the history and proves nothing.
   640	        active.route.observeFinal(active.record)
   641	        try {
   642	            // stop() unblocks a pending read. Do not release here while the reader may
   643	            // still be using the same AudioRecord instance.
   644	            active.record.stop()
   645	        } catch (e: Exception) {
   646	            // The reader's finally block still owns and releases the resources if stop
   647	            // itself fails, so a vendor-specific AudioRecord error cannot leak a session.
   648	            DebugLogger.warn(TAG, "AudioRecord stop failed: ${e.message}")
   649	        }
   650	        DebugLogger.mark(TAG, "recording_stop")
   651	        // The ending is read back from the CLAIM rather than from this function's parameter, because the
   652	        // claim is the owner of the answer and the parameter is only what this caller proposed.
   653	        //
   654	        // Naming it here is what makes a stop button and a silence stop tell apart in a log at all. The
   655	        // other two endings already write their own distinct lines from the capture loop; these two wrote
   656	        // the same one. Ordering against the foreground-service line is NOT a discriminator, because
   657	        // `claimEnding` clears `isRecording` before this log runs, so the session's polling thread can log
   658	        // its own promotion first. Measured on the S26 and recorded in issue #114: a silence-ended take
   659	        // did exactly that, by 11 ms.
   660	        DebugLogger.log(
   661	            TAG,
   662	            "Stopped by ${active.endingClaim.ending.label}. ${active.bytesWritten} bytes " +
   663	                "(${String.format("%.1f", PcmAudio.durationSeconds(active.bytesWritten))}s) -> ${active.file.absolutePath}",
   664	        )
   665	    }

   667	    private fun releaseSession(active: CaptureSession) {
   668	        var holding = false
   669	        synchronized(sessionLock) {
   670	            if (session !== active) return
   671	            active.route.stopWatching()
   672	            // Capture-loop endings (cap, byte ceiling, error) reach here with the recorder still active.
   673	            active.route.observeFinal(active.record)
   674	            // The one handoff point every ending reaches: a manual stop, the silence stop and both caps
   675	            // may keep the earbuds warm; an error ending and teardown release everything.
   676	            holding = warmHoldOwner.eligible(active.route, active.endingClaim.ending, active.keepEarbudsReady, destroyed) &&
   677	                warmHoldOwner.start(active.route)
   678	            closeResources(active, keepRoute = holding)
   679	            lastSilenceStatus = active.detector.status
   680	            session = null
   681	            if (captureThread === Thread.currentThread()) captureThread = null
   682	            currentAmplitude = 0f
   683	        }
   684	
   685	        // Audio and the PCM file are already closed above. Detector cleanup therefore cannot delay the
   686	        // file becoming ready, which is what the user is waiting for. Nothing here blocks: each owner's
   687	        // close tells its thread to stop and abandons it (#188); the feeder unbinds as it exits.
   688	        active.detector.close(unbindNow = false)
   689	        active.picture.close()
   690	        // Once per take, on EVERY ending (a stop, a silence stop, a cap, a capture error, teardown):
   691	        // release is the one point they all reach. Shape only; `polled` is the proof that no production
   692	        // code polls the picture any more (#187).
   693	        DebugLogger.log(TAG, "Live picture: pushed=${active.picture.pushes.get()} polled=${active.picture.polls.get()}")
   694	        // A hold keeps the service alive; its end calls stopSelf (RULE: the service owns its own end).
   695	        if (!holding) stopSelf()
   696	    }
   697	
   698	    /**
   699	     * The route is released HERE, synchronously, before the session slot frees (the caller clears
   700	     * `session` after this returns under `sessionLock`), so a later start can never install a request
   701	     * that an earlier release still has to clear. The loop is over, so the binder call delays no read.
   702	     *
   703	     * With [keepRoute] only the recorder's listener goes (it dies with the `AudioRecord`); the
   704	     * communication ownership stays in the `RouteHold` the warm hold now carries.
   705	     */
   706	    private fun closeResources(active: CaptureSession, keepRoute: Boolean) {
   707	        active.route.close(keepRoute)
   708	        closeResources(active.record, active.output)
   709	    }
   710	
   711	    private fun closeResources(record: AudioRecord?, output: FileOutputStream?) {
   712	        runCatching { output?.flush() }
   713	            .onFailure { DebugLogger.warn(TAG, "Failed to flush audio file: ${it.message}") }
   714	        runCatching { output?.close() }
   715	            .onFailure { DebugLogger.warn(TAG, "Failed to close audio file: ${it.message}") }
   716	        runCatching { record?.stop() }
   717	            .onFailure {
   718	                if (it !is IllegalStateException) DebugLogger.warn(TAG, "Failed to stop AudioRecord: ${it.message}")
   719	            }
   720	        runCatching { record?.release() }
   721	            .onFailure { DebugLogger.warn(TAG, "Failed to release AudioRecord: ${it.message}") }
   722	    }

   724	    /** Wait for the capture thread to finish writing and close the file. */
   725	    fun waitForFileReady(timeoutMs: Long = 2_000L): Boolean {
   726	        val thread = captureThread
   727	        if (thread != null && Thread.currentThread() !== thread) {
   728	            thread.join(timeoutMs.coerceAtLeast(0L))
   729	        }
   730	        val ready = thread?.isAlive != true && session == null
   731	        if (ready) captureThread = null
   732	        return ready
   733	    }

   735	    override fun onDestroy() {
   736	        // Ordered: no hold may start after this flag, so the take stopRecording ends below cannot open
   737	        // one after the route thread is gone (Codex review 1).
   738	        destroyed = true
   739	        synchronized(sessionLock) { warmHoldOwner.close(WarmHold.END_DESTROYED) }
   740	        stopRecording()
   741	        session?.let { active ->
   742	            active.detector.close(unbindNow = true)
   743	            active.picture.close()
   744	        }
   745	        val thread = captureThread
   746	        if (thread != null && thread !== Thread.currentThread()) {
   747	            thread.join(2_000L)
   748	            if (thread.isAlive) {
   749	                // The capture thread is the sole owner of AudioRecord and the file. Do not
   750	                // close either resource here after the bounded wait. Process termination of
   751	                // the isolated :audio service will reclaim them without an ANR-length wait.
   752	                DebugLogger.warn(TAG, "Capture thread did not finish during service teardown")
   753	            }
   754	        }
   755	        if (captureThread?.isAlive != true) {
   756	            captureThread = null
   757	            synchronized(sessionLock) {
   758	                isRecording.set(false)
   759	            }
   760	        }
   761	        // A hold that slipped in between the flag and the join is ended here, before its expiry dies.
   762	        synchronized(sessionLock) { warmHoldOwner.close(WarmHold.END_DESTROYED) }
   763	        // After the join: the capture thread's cleanup removed its listener; nothing else posts here.
   764	        routeThread.quitSafely()
   765	        super.onDestroy()
   766	    }

    14	/**
    15	 * The silence detector's feed for ONE take: the ring the capture thread stages whole blocks into, the
    16	 * binding to the detector process, the one thread allowed to talk to it, and the status the binder reads.
    17	 *
    18	 * Owned by the take's `CaptureSession`, started after capture is up, closed exactly once by the service
    19	 * (#188). Everything here belongs to the take that started it, so a feeder or a connection callback
    20	 * belonging to a finished take cannot set the status of, or unbind the detector of, the take running now:
    21	 * the service's identity checks come in as [isCurrent] and [stillLive], and the one decision that
    22	 * crosses ownership, ending the take on silence, as [endOnSilence].
    23	 *
    24	 * [bind], [unbind] and [interrupt] are the platform edges, injected so a JVM test can count that [close]
    25	 * runs each once. Null [ring] means auto-stop is off: no ring, no feeder, no detector process.
    26	 */
    27	internal class DetectorFeed(
    28	    private val tag: String,
    29	    autoStop: Boolean,
    30	    private val bind: (ServiceConnection) -> Boolean,
    31	    private val unbind: (ServiceConnection) -> Unit,
    32	    private val interrupt: (Thread) -> Unit = { it.interrupt() },
    33	) {
    34	    companion object {
    35	        /**
    36	         * The silence detector's block: 4096 samples, 256 ms at 16 kHz, macOS's detector chunk and what
    37	         * the silence state machine ticks on. Reads are staged into whole blocks; the detector never
    38	         * sees a partial one.
    39	         */
    40	        const val READ_BLOCK_BYTES = 8_192
    41	
    42	        /**
    43	         * Eight blocks, 2.048 seconds of audio, and the detector's own call deadline is set against it.
    44	         * It is the client-owned deadline for the one binding failure Android gives no signal for: a
    45	         * bind that succeeds and then never connects.
    46	         */
    47	        const val RING_BLOCKS = 8
    48	
    49	        /** How long the feeder waits when there is nothing to do. It is not the capture thread. */
    50	        const val FEEDER_IDLE_MS = 20L
    51	
    52	        /** The production [bind]: the detector process, bound with auto-create. */
    53	        fun bindingThrough(context: Context): (ServiceConnection) -> Boolean = { connection ->
    54	            context.bindService(Intent(context, SilenceVadService::class.java), connection, Context.BIND_AUTO_CREATE)
    55	        }
    56	
    57	        /** The production [unbind]. */
    58	        fun unbindingThrough(context: Context): (ServiceConnection) -> Unit = { connection -> context.unbindService(connection) }
    59	    }
    60	
    61	    /** Null when the user has auto-stop off: no ring, no feeder, no detector process. */
    62	    private val ring: BlockRing? = if (autoStop) BlockRing(RING_BLOCKS, READ_BLOCK_BYTES) else null
    63	
    64	    /** Staging for a read that did not land on a block boundary. Preallocated, like everything else. */
    65	    private val pendingBlock: ByteArray? = if (autoStop) ByteArray(READ_BLOCK_BYTES) else null
    66	
    67	    /** Capture thread only. */
    68	    private var pendingBytes: Int = 0
    69

    71	    private var pendingPosition: Long = 0L
    72	
    73	    private val detectorAbandoned = AtomicBoolean(false)
    74	    private val silenceStatus = AtomicInteger(
    75	        if (ring == null) AudioCaptureService.SILENCE_STATUS_DISABLED else AudioCaptureService.SILENCE_STATUS_PREPARING,
    76	    )
    77	    @Volatile private var vadService: ISilenceVadService? = null
    78	    /** Exactly one unbind per bind: the feeder's exit and a teardown close can both reach [unbindVad]. */
    79	    private val vadBound = AtomicBoolean(false)
    80	    @Volatile private var feederThread: Thread? = null
    81	    @Volatile private var vadConnection: ServiceConnection? = null
    82	    private val closed = AtomicBoolean(false)
    83	
    84	    /** The detector's status for the binder, one of the `SILENCE_STATUS_*` codes. */
    85	    val status: Int get() = silenceStatus.get()
    86	
    87	    /** The caller asked for auto-stop and cannot have it (an out-of-range pause): the notice's state. */
    88	    fun markRequestedButRefused() {
    89	        silenceStatus.set(AudioCaptureService.SILENCE_STATUS_UNAVAILABLE)
    90	    }
    91	
    92	    /**
    93	     * Capture thread only. Copies audio toward the detector and never waits for it.
    94	     *
    95	     * A read that does not land on a block boundary is staged, so the detector always sees whole 256 ms
    96	     * blocks in order. **Nothing here logs, allocates, locks or calls across a process.**
    97	     *
    98	     * A full ring means the detector has fallen further behind than it can recover from. Auto-stop is
    99	     * abandoned for the rest of the take and never resumed: resuming across dropped audio breaks the
   100	     * model's recurrent continuity, and speech that resumed inside the gap could then read as silence.
   101	     */
   102	    fun offer(buffer: ByteArray, bytesRead: Int, position: Long) {
   103	        val ring = ring ?: return
   104	        val pending = pendingBlock ?: return
   105	        if (detectorAbandoned.get()) return
   106	
   107	        var consumed = 0
   108	        while (consumed < bytesRead) {
   109	            if (pendingBytes == 0) pendingPosition = position + consumed
   110	            val room = READ_BLOCK_BYTES - pendingBytes
   111	            val take = minOf(room, bytesRead - consumed)
   112	            System.arraycopy(buffer, consumed, pending, pendingBytes, take)
   113	            pendingBytes += take
   114	            consumed += take
   115	            if (pendingBytes == READ_BLOCK_BYTES) {
   116	                pendingBytes = 0
   117	                if (!ring.offer(pending, READ_BLOCK_BYTES, pendingPosition)) {
   118	                    // Flag only. The feeder notices and does the logging, off this thread.
   119	                    abandon()
   120	                    return
   121	                }
   122	            }
   123	        }
   124	    }

   126	    /**
   127	     * Every way the binding can fail, and they all mean the same thing to a take: auto-stop is off for
   128	     * it, and recording continues.
   129	     *
   130	     * The connection is built PER TAKE and captures this feed. A callback that arrives after its take
   131	     * ended can then do nothing at all, rather than clearing the status or unbinding the detector of
   132	     * whatever is recording now. `onNullBinding` and `onBindingDied` unbind explicitly, because Android
   133	     * reconnects a disconnected binding on its own.
   134	     */
   135	    private fun connectionFor(isCurrent: () -> Boolean) = object : ServiceConnection {
   136	        override fun onServiceConnected(name: ComponentName?, binder: IBinder?) {

   147	        }

   162	    /**
   163	     * Bind the detector process for one take and start the one thread allowed to talk to it.
   164	     *
   165	     * Capture has already started by the time this runs, so a slow or failed detector delays nothing. A
   166	     * take whose detector never becomes ready is simply a take the user stops by hand.
   167	     */
   168	    fun start(
   169	        pauseSeconds: Float,
   170	        token: Long,
   171	        takeId: String,
   172	        isCurrent: () -> Boolean,
   173	        stillLive: () -> Boolean,
   174	        endOnSilence: () -> Unit,
   175	    ) {
   176	        val connection = connectionFor(isCurrent)
   177	        vadConnection = connection
   178	
   179	        val bound = runCatching { bind(connection) }.getOrDefault(false)
   180	        vadBound.set(bound)
   181	
   182	        if (!bound) {
   183	            vadConnection = null
   184	            abandon()
   185	            DebugLogger.warn(tag, "Auto-stop unavailable: detector service could not be bound")
   186	            return
   187	        }
   188	
   189	        val thread = Thread({ feederLoop(pauseSeconds, token, takeId, stillLive, endOnSilence) }, "SilenceFeederThread")
   190	        feederThread = thread
   191	        runCatching { thread.start() }
   192	            .onFailure {
   193	                feederThread = null
   194	                abandon()
   195	                unbindVad()
   196	                DebugLogger.warn(tag, "Auto-stop unavailable: detector feeder could not start")
   197	            }
   198	    }
   199	
   200	    /**
   201	     * The only thread that calls the detector. It is allowed to block; the capture thread is not.
   202	     *
   203	     * The abandonment flag is checked at the top of every pass AND immediately after every remote call,
   204	     * because a gap can open while a call is in flight and a verdict computed from the blocks before a
   205	     * gap must never be applied to the audio after it.
   206	     */
   207	    private fun feederLoop(
   208	        pauseSeconds: Float,
   209	        token: Long,
   210	        takeId: String,
   211	        stillLive: () -> Boolean,
   212	        endOnSilence: () -> Unit,
   213	    ) {
   214	        val ring = ring ?: return
   215	        val block = ByteArray(READ_BLOCK_BYTES)
   216	        var started = false
   217	        var reportedAbandon = false
   218	
   219	        fun shouldStop(): Boolean {
   220	            if (detectorAbandoned.get()) {
   221	                if (!reportedAbandon) {
   222	                    reportedAbandon = true
   223	                    DebugLogger.warn(tag, "Auto-stop abandoned for this take")
   224	                }
   225	                return true
   226	            }
   227	            return !stillLive()
   228	        }
   229	
   230	        try {
   231	            while (!shouldStop()) {
   232	                val remote = vadService
   233	                if (remote == null) {
   234	                    Thread.sleep(FEEDER_IDLE_MS)
   235	                    continue
   236	                }
   237	
   238	                if (!started) {
   239	                    // The versioned start carries the take's id as detector context; a legacy start (no
   240	                    // id) keeps the old transaction (issue #176).
   241	                    val status = runCatching {
   242	                        if (takeId.isEmpty()) remote.start(token, pauseSeconds)
   243	                        else remote.startForTake(token, pauseSeconds, takeId)
   244	                    }
   245	                        .getOrElse {
   246	                            abandon()
   247	                            DebugLogger.warn(tag, "Auto-stop unavailable: start failed, ${it.message}")
   248	                            return
   249	                        }
   250	                    if (shouldStop()) return
   251	                    if (status != SilenceVadService.STATUS_READY) {
   252	                        abandon()
   253	                        DebugLogger.warn(tag, "Auto-stop unavailable: the detector reported so")
   254	                        return
   255	                    }
   256	                    started = true
   257	                    silenceStatus.compareAndSet(
   258	                        AudioCaptureService.SILENCE_STATUS_PREPARING,
   259	                        AudioCaptureService.SILENCE_STATUS_READY,
   260	                    )
   261	                }
   262	
   263	                val length = ring.poll(block)
   264	                if (length <= 0) {
   265	                    Thread.sleep(FEEDER_IDLE_MS)
   266	                    continue
   267	                }
   268	
   269	                val result = runCatching { remote.processBlock(token, block) }
   270	                    .getOrElse {
   271	                        abandon()
   272	                        DebugLogger.warn(tag, "Auto-stop unavailable: the detector call failed")
   273	                        return
   274	                    }
   275	
   276	                if (shouldStop()) return
   277	
   278	                when (result) {
   279	                    SilenceVadService.RESULT_SILENCE -> {
   280	                        // endTake re-checks that this session is still the live one, under the lock, so
   281	                        // a verdict from a finished take cannot end the take running now.
   282	                        endOnSilence()
   283	                        return
   284	                    }
   285	
   286	                    SilenceVadService.RESULT_UNAVAILABLE -> {
   287	                        abandon()
   288	                        DebugLogger.warn(tag, "Auto-stop unavailable: the detector gave up mid-take")
   289	                        return
   290	                    }
   291	                }
   292	            }
   293	        } catch (interrupted: InterruptedException) {
   294	            Thread.currentThread().interrupt()
   295	        } catch (e: Exception) {
   296	            abandon()
   297	            DebugLogger.warn(tag, "Auto-stop unavailable: the feeder failed, ${e.message}")
   298	        } finally {
   299	            if (started) runCatching { vadService?.finish(token) }
   300	            unbindVad()
   301	            if (feederThread === Thread.currentThread()) feederThread = null
   302	        }
   303	    }
   304	
   305	    /**
   306	     * Auto-stop is off for the rest of THIS take, and is never resumed within it.
   307	     *
   308	     * The status it lands on records whether the detector ever worked. A take that never got one tells
   309	     * the user; a take that had one and lost it does not, because that recording is still correct and a
   310	     * message part way through is an interruption for nothing.
   311	     */
   312	    fun abandon() {
   313	        detectorAbandoned.set(true)
   314	        while (true) {
   315	            val previous = silenceStatus.get()
   316	            val next = when (previous) {
   317	                AudioCaptureService.SILENCE_STATUS_DISABLED,
   318	                AudioCaptureService.SILENCE_STATUS_UNAVAILABLE,
   319	                AudioCaptureService.SILENCE_STATUS_LOST_AFTER_READY -> return
   320	
   321	                AudioCaptureService.SILENCE_STATUS_READY -> AudioCaptureService.SILENCE_STATUS_LOST_AFTER_READY
   322	                else -> AudioCaptureService.SILENCE_STATUS_UNAVAILABLE
   323	            }
   324	            if (silenceStatus.compareAndSet(previous, next)) return
   325	        }
   326	    }
   327	
   328	    /** Unbinds only this feed's own connection, so a finished take cannot unbind a running one's. */
   329	    private fun unbindVad() {
   330	        if (!vadBound.compareAndSet(true, false)) return
   331	        val connection = vadConnection ?: return
   332	        vadConnection = null
   333	        vadService = null
   334	        runCatching { unbind(connection) }
   335	            .onFailure { DebugLogger.warn(tag, "Detector unbind failed: ${it.message}") }
   336	    }
   337	
   338	    /**
   339	     * Abandon and interrupt the feeder. Idempotent. Nothing here blocks: the feeder is told to stop and
   340	     * abandoned, and it holds no recorder, no stream, no ring slot and no reference to a later take.
   341	     * Read [status] BEFORE calling this, because abandoning can move it.
   342	     *
   343	     * The unbind is the feeder's, in its `finally`, on an ordinary release: the take's file is already
   344	     * closed and nothing waits on the detector. [unbindNow] is for teardown, where the process may not
   345	     * give the feeder another turn; [unbindVad] admits exactly one of the two.
   346	     */
   347	    fun close(unbindNow: Boolean) {
   348	        if (closed.compareAndSet(false, true)) {
   349	            detectorAbandoned.set(true)
   350	            feederThread?.let { runCatching { interrupt(it) } }
   351	        }
   352	        if (unbindNow) runCatching { unbindVad() }
   353	    }
   354	}

   273	     * miss: reset the communication device once, if the sink is still there. Second miss: proceed without
   274	     * sound on the earbuds (FORCED), or, when the observed route is the phone with earbuds connected,
   275	     * fail the take rather than record from the phone.

   291	                    when (gate.deadlinePassed()) {

   298	                        }

   300	                            deadline = null
   301	                            if (admissible()) {
   302	                                gate.force()

   307	                                onRefused()
   308	                            }
   309	                        }
   310	                    }
   311	                }
   312	            }
   313	        }
   314	        deadline = runnable
   315	        scheduler.postDelayed(runnable, LiveGate.DEADLINE_MS)
   316	    }

   321	        val stillThere = runCatching {
   322	            audioManager.availableCommunicationDevices.any { it.type == sink.type && it.productName?.toString() == sink.productName?.toString() }
   323	        }.getOrDefault(false)
   324	        if (!stillThere) return false

   326	            audioManager.clearCommunicationDevice()
   327	            audioManager.setCommunicationDevice(sink).also { if (it) hold.markCommunicationSet() }
   328	        }.getOrElse { e ->
   329	            DebugLogger.warn(tag, "communication device reset threw: ${e.message}")
   330	            false
   331	        }
   332	    }
   333

   336	     * deadline or removal callback then finds nothing to do. Idempotent; [close] runs it again.
   337	     */
   338	    fun stopWatching() {
   339	        deadline?.let { runCatching { scheduler.removeCallbacks(it) } }
   340	        deadline = null
   341	        sinkWatch?.let { runCatching { unregisterDeviceCallback(it) } }
   342	        sinkWatch = null
   343	    }

   347	     * so a later start can never install a request that an earlier release still has to clear. With

   355	    }

     6	    const val BYTES_PER_SAMPLE = 2

     8	    /**
     9	     * How much audio one `AudioRecord.read` asks for: 512 samples, 32 ms at 16 kHz.
    10	     *
    11	     * **This is a different quantity from the buffer the AudioRecord is constructed with**, and the
    12	     * two want opposite things. The native buffer is the margin that stops an overrun when the
    13	     * capture thread is descheduled, so it wants to be large. This is the loop's decision
    14	     * granularity, so it wants to be small: the duration ceiling can only fire on a read boundary,
    15	     * and so can a silence stop. Reading the whole native buffer made both coarse to about a second.
    16	     *
    17	     * Android's own guidance is to read in short frequent chunks rather than waiting for the buffer
    18	     * to fill. 32 ms is what the recorder's live picture needs: a syllable is about 100 ms, and the
    19	     * 256 ms read this replaced handed the meter one averaged number per quarter second, which
    20	     * cannot show a voice (#151). The capture service reads with it and the picture sizes its ring
    21	     * from it (#188).
    22	     */
    23	    internal const val READ_CHUNK_BYTES = 1_024
    24	
    25	    fun durationSeconds(byteCount: Long): Float =
    26	        byteCount.toFloat() / (SAMPLE_RATE * BYTES_PER_SAMPLE)
    27	
    28	    /** Decode complete little-endian PCM16 samples. A trailing byte is ignored. */
    29	    fun toFloatSamples(pcmData: ByteArray): FloatArray {
    30	        val samples = FloatArray(pcmData.size / BYTES_PER_SAMPLE)
    31	        for (i in samples.indices) {
    32	            val offset = i * BYTES_PER_SAMPLE
    33	            val value = (pcmData[offset].toInt() and 0xFF) or
    34	                (pcmData[offset + 1].toInt() shl 8)
    35	            samples[i] = value / 32768.0f
    36	        }
    37	        return samples
    38	    }
